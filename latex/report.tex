\section{The OU Process}
    \begin{enumerate}
        \item 
            \begin{gather*}
                dX_t = \theta(\mu-X_t)dt + \sigma dW_t, \quad \theta, \sigma > 0
                \\
                d(e^{\theta t}X_t) = e^{\theta t}dX_t + \theta e^{\theta t}X_tdt
                \\
                d(e^{\theta t}X_t) =  e^{\theta t}\left(\theta(\mu-X_t)dt +
                \sigma dW_t\right) + \theta e^{\theta t}X_t dt
                \\
                d(e^{\theta t}X_t) = \theta e^{\theta t}\mu dt +
                \sigma e^{\theta t} dW_t
                \\
                e^{\theta t}X_t - X_0 = \mu (e^{\theta t} - 1) + \sigma \int_0^t e^{\theta s}dW_s
                \\
                X_t = \mu - (X_0 - \mu)e^{-\theta t} + \sigma \int_0^t e^{-\theta
                (t - s)}dW_s
            \end{gather*}
        \item 
            We begin by investigating the stochastic term in the previous
            result: 
            \begin{gather*}
                X_t = \alpha + \beta, \quad \alpha = \mu - (X_0 - \mu)e^{-\theta
                t}, \quad \beta = \sigma \int_0^t e^{-\theta (t - s)}dW_s
                \\
                \beta \sim N(0,
                \sigma^2\int_0^t e^{-2\theta (t - s)}ds)
                \\
                \sim N\left(0, \frac{\sigma^2}{2\theta}\left[1 -  e^{-2\theta t}\right]\right)
            \end{gather*}
            \begin{gather*}
                E(X_t) = E\left(\mu - (X_0 - \mu)e^{-\theta t} + \sigma \int_0^t e^{-\theta
                (t - s)}dW_s\right)
                \\
                = \mu - (X_0 - \mu)e^{-\theta t} + E(\sigma \int_0^t e^{-\theta
                (t - s)}dW_s) 
                \\
                = \mu - (X_0 - \mu)e^{-\theta t}
            \end{gather*} 
            \begin{gather*} 
                \Var(X_t) = E(X_t^2) - E(X_t)^2
                \\
                = E\left(\alpha^2 + 2\alpha\beta + \beta^2 \right)- \alpha^2
                \\
                =  E(\beta^2)
                \\
                = \frac{\sigma^2}{2\theta}\left[1 -  e^{-2\theta t}\right]
            \end{gather*}
            This implies that the stationary distribution for $\lim_{t\to\infty}
            X_t$ is 
            \begin{gather*}
                \lim_{t\to\infty} X_t \sim N\left(\mu,
                \frac{\sigma^2}{2\theta}\right)
            \end{gather*}
        \item \textbf{Half-lfe of a deviation} First we define a deviation as a
        perturbation from an initial condition. Since this is a stochastic
        timeseries, we can inject this deviation at any point in the timeseries.
        We define its half-life $\lambda$ as how long it takes for that deviation to
        decrease to half of its original value. 

        \begin{gather*}
            X_t | X_O + d = \mu + (X_0 + d - \mu) e^{\theta t} + \sigma
            \int_0^{t} e^{-\theta (t - s)}dW_s
            \\
            X_t | X_O + d = \left(X_t | X_0\right) + d e^{-\theta t}
        \end{gather*}
        We find that the deviation follows an expoential decay curve given by
        $e^{-\theta t}$. The half life of this is easy to measure, we have
        \begin{gather*}
            \frac{1}{2} = e^{-\theta \lambda}
            \\
            \lambda = \frac{\ln(2)}{\theta}
        \end{gather*}

        \item \textbf{Exact Discretization}: 
            We find the exact discretization by chosing $t = \Delta t$ in the original
            solution to the SDE. We find
            \begin{gather*}
                X_{n+1} | X_{n} = \mu + (X_{n} - \mu)e^{-\Delta t \theta} +
                \int_0^{\Delta t} e^{-\theta (\Delta t - s)}dW_s
                \\
                = \mu\left(1 - e^{-\Delta t\theta}\right) + X_n e^{-\Delta t \theta} + \int_0^{\Delta t} e^{-\theta (\Delta t - s)}dW_s
                \\
                = a + bX_n + \epsilon_k
            \end{gather*}
            Here we have shown that $a = \mu(1 - b)$, $b
            = e^{-\Delta t \theta}$ and $\epsilon_k = \int_0^{\Delta t}
            e^{-\theta (\Delta t - s)}dW_s$. We have already shown (using
            $\Delta t$ to replace $t$) that $\epsilon_k \sim
            N\left(0,\frac{\sigma^2}{2\theta}(1 - b^2)\right)$. We invert these
            equations to find, 
            \begin{gather*}
                \theta =- \frac{\ln(b)}{\Delta t}
                \\
                \mu = \frac{a}{1 - b}
                \\
                \sigma = -\frac{2\Var(\epsilon_k)\ln(b)}{\Delta t(1 - b^2)}
            \end{gather*}
    \end{enumerate}

        \item \textbf{Maximum-Likelihood Estimation}
        Derive MLE of $(\theta, \mu, \sigma)$. Show it is equivalent to
        the least squares regression of $X_{k+1}$ on $X_{k}$. 

        We begin by identifying the probability/likelihood of $X_{n+1}$ given
        $X_{n}$ which is gaussian. In order to simplify the equation, we denote $\mu_{\star} =
        E(X_{n+1} | X_n)$ and $\sigma_{\star} = \Var(X_{n+1} | X_n)$.
        \begin{gather*}
            P_n = P(X_{n+1} | X_{n}; \theta, \mu, \sigma) =
            \frac{1}{\sigma_{\star}\sqrt{2\pi}}\exp\left(-\frac{(X_{n+1} -
            \mu_{\star})^2}{2\sigma_{\star}^2}\right)
        \end{gather*}
        Thus we can assemble the (log) likelihood function $L$ by taking the product
        of these possibilies for $X_i$ for all $i \in [0, N_t]$. We have, 
        \begin{gather*}
            L(\bs{X}_{n+1} | \bs{X}_{n}; \theta, \mu, \sigma) =
            \prod_{i=0}^{N_t-1}P_i
            \\
            \ln(L) = \sum_{i=0}^{N_t -1} \ln(P_i)
            \\
            \ln(L) = -N_t\ln(\sigma_{\star}\sqrt{2\pi}) - \sum_{i}
            \frac{(X_{n+1} - 
            \mu_{\star})^2}{2\sigma_{\star}^2}
        \end{gather*}
        Without loss of generality, we will study the maximum likelihood of
        $X_1|X_0$ as the inclusion of additional terms will only add rows to the
        eventual linear regression problem. 
        \begin{gather*}
            \mathcal{L} = \ln(L(X_1|X_0;\theta, \mu, \star)) = 
            -\ln(\sigma_{\star}\sqrt{2\pi}) - 
            \frac{(X_{1} - 
            \mu_{\star})^2}{2\sigma_{\star}^2}
            \\
            \mathcal{L}_{\mu_{\star}} = \frac{(X_1 -
            \mu_{\star})}{\sigma_{\star}^2} = 0 
            \\
            \mathcal{L}_{\sigma_{\star}} = -\frac{1}{\sigma_{\star}} +
            \frac{(X_1 - \mu_{\star})^2}{\sigma_{\star}^3} = 0 
        \end{gather*}
        From these two partial derivatives of the log-likelihood function we
        can solve for $\theta, \mu, \sigma$. From the partial
        $\mathcal{L}_{\mu_{\star}}$, we find
        \begin{gather*}
            \frac{(X_1 -
            \mu_{\star})}{\sigma_{\star}^2} = 0
            \\
            X_1 - \mu - (X_0 - \mu) e^{-\Delta t \theta} = 0
            \\
            X_1 = X_0e^{-\Delta t \theta} - \mu(1 - e^{\Delta t \theta})
        \end{gather*}
        This reduces the maximum-likelihood estimation for $\theta, \mu$ to a
        linear regression from $X_0$ to $X_1$, i.e. we obtain $(a,b)$ from the
        linear update function. The slope and y-intercept of that regression
        yield $\theta$ and $\mu$. 

        Next we look at the equation for $\mathcal{L}_{\sigma_{\star}}$: 
        \begin{gather*}
            \frac{(X_1 -
            \mu_{\star})^2}{\sigma_{\star}^3} = \frac{1}{\sigma_{\star}}
            \\
            \left(X_1 - X_0e^{-\Delta t \theta} - \mu(1 - e^{\Delta t
            \theta})\right)^2 = \sigma_{\star}^2
        \end{gather*}
        From this equation, we find that square of the residual for this linear
        fit is exactly equal to the ``modified" variance for this update
        probability. To generalize this for $N_t$ timesteps, we turn the linear
        regression into a least squares regression problem where we fit a single
        $a$ and $b$ (c.f. problem 1.4) to fit each timestep update. Most modern
        Least-Squares solvers return the sum of the squared residuals as an
        output alongside the linear regression. In order to compute the
        ``modified" variance $(\sim \Var(\epsilon_k))$, we divide the the sum of
        squared residuals by $N_t-1$. We now have $(a, b, \Var(\epsilon_k))$, and
        can invert the equations to solve for $(\theta, \mu, \sigma)$.

    \section{Time Varying Hedge Ratio}
    \begin{enumerate}
        \item \textbf{Kalman-Filter}
        We begin by writing the state-space model,
        \begin{gather*}
            \ln(P_A) = \alpha_t + \beta_t\ln(P_B) + v_t
        \end{gather*}
        where $\alpha_t$ and $\beta_t$ are described by random walks and $v_t$
        is a zero-mean gaussian with variance $R$ describing some noise in the
        price data. 
        We will now look at the evolution of $(\alpha_t,\beta_t)$. We know that
        increments to these are i.i.d. white noise. Thus, 
        \begin{gather*}
            \left[\begin{array}{c} \alpha_{t}\\\beta_t\end{array}\right] = 
            \left[\begin{array}{c} \alpha_{t-1}\\\beta_{t-1}\end{array}\right] +
            \bs{dW}
        \end{gather*}
        Thus we expect noise to be injected into the state model with variance $\Delta t$. Let us
        assume that we have already done the task of fitting $(\alpha_t,
        \beta_t)$ for some $t \le t_0$ with confidence $P$. We would then have the
        following prediction for $t_1$:
        \begin{gather*}
            \hbs{\alpha}_t = (\hat{\alpha}_t, \hat{\beta}_t) = (\alpha_{t-1}, \beta_{t-1})
            \hbs{P}_t = \bs{P}_{t-1} + Q
        \end{gather*}
        We then look at how closely these $\alpha_t, \beta_t$ fit the
        observation model: $\ln(P_A) = \bs{H}_t\cdot\bs{\alpha}_t$. 
        We obtain the error in our observations given by: 
        \begin{gather*}
            e_t = \ln(P_A) - \bs{H}_t\cdot\hbs{\alpha}_t
        \end{gather*}
        This is coupled with the uncertainty/confidence of our our state model
        being injected into our observations. We find the confidence at this
        level being, 
        \begin{gather*}
            oe_t = \bs{H}_t\cdot(\hbs{P}_t\bs{H}_t) + R
        \end{gather*}
        and finally the ratio between our state and observation confidence being
        equivalent to the kalman gain $K_t$. 
        \begin{gather*}
            \bs{K}_t = \frac{1}{oe_t}\hbs{P}_t\bs{H}_t
        \end{gather*}
        Using the Kalman gain we can correct our model, we have
        \begin{gather*}
            \bs{\alpha}_t = \hbs{\alpha}_t + \bs{K}_te_t
            \\
            \bs{P}_t = (I - \bs{K}_t\bs{H}_t^T)\hbs{P}_t
        \end{gather*}
        Therefore we have a nice recursion, if we know our Kalman filter
        $\bs{\alpha}_t$ and $\bs{P}_t$ we can always update the filter with
        another timestep following the above scheme. Doing this over, the
        entirety of the given timeseries allows us to find a hedge ratio
        $\beta_t$ which is closer to the true hedge ratio at a given time $t$.
        Note that that Kalman filter can be updated as new data is
        presented and therefore so can our spread. As new data is introduced to
        our system, after updating the hedge-ratio/spread, we have a new
        timestep to fit the SDE. A simple
        recursive least-squares solution is performed to update $\mu, \theta,
        \sigma$.
