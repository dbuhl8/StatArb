    \begin{enumerate}
        \item 
            \begin{gather*}
                dX_t = \theta(\mu-X_t)dt + \sigma dW_t, \quad \theta, \sigma > 0
                \\
                d(e^{\theta t}X_t) = e^{\theta t}dX_t + \theta e^{\theta t}X_tdt
                \\
                d(e^{\theta t}X_t) =  e^{\theta t}\left(\theta(\mu-X_t)dt +
                \sigma dW_t\right) + \theta e^{\theta t}X_t
                \\
                d(e^{\theta t}X_t) = \theta e^{\theta t}\mu dt +
                \sigma e^{\theta t} dW_t
                \\
                e^{\theta t}X_t - X_0 = \mu (e^{\theta t} - 1) + \sigma \int_0^t e^{\theta s}dW_s
                \\
                X_t = \mu - (X_0 - \mu)e^{-\theta t} + \sigma \int_0^t e^{-\theta
                (t - s)}dW_s
            \end{gather*}
        \item 
            We begin by investigating the stochastic term in the previous
            result: 
            \begin{gather*}
                X_t = \alpha + \beta, \quad \alpha = \mu - (X_0 - \mu)e^{-\theta
                t}, \quad \beta = \sigma \int_0^t e^{-\theta (t - s)}dW_s
                \sigma \int_0^t e^{-\theta (t - s)}dW_s \sim N(0,
                \sigma^2\int_0^t e^{-2\theta (t - s)}ds)
                \\
                \sim N\left(0, \frac{\sigma^2}{2\theta}\left[1 -  e^{-2\theta t}\right]\right)
            \end{gather*}
            \begin{gather*}
                E(X_t) = E\left(\mu - (X_0 - \mu)e^{-\theta t} + \sigma \int_0^t e^{-\theta
                (t - s)}dW_s\right)
                \\
                = \mu - (X_0 - \mu)e^{-\theta t} + E(\sigma \int_0^t e^{-\theta
                (t - s)}dW_s) 
                \\
                = \mu - (X_0 - \mu)e^{-\theta t}
            \end{gather*} \begin{gather*} X_t 
                Var(X_t) = E(X_t^2) - E(X_t)^2
                \\
                = E\left(\alpha^2 + 2\alpha\beta + \beta^2 \right)- \alpha^2
                \\
                =  E(\beta^2)
                \\
                = \frac{\sigma^2}{2\theta}\left[1 -  e^{-2\theta t}\right]
            \end{gather*}
            This implies that the stationary distribution for $\lim_{t\to\infty}
            X_t$ is 
            \begin{gather*}
                \lim_{t\to\infty} X_t \sim N\left(\mu,
                \frac{\sigma^2}{2\theta}\right)
            \end{gather*}

\end{enumerate}
